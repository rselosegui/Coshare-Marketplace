const FilterModal = ({ filters, onUpdate, onClose }: { filters: FilterState, onUpdate: (f: FilterState) => void, onClose: () => void }) => {
  const [localFilters, setLocalFilters] = useState<FilterState>(filters);

  return (
    <div 
      onClick={onClose}
      className="absolute inset-0 z-50 flex items-end justify-center px-4 bg-primary dark:bg-gray-900/20 backdrop-blur-sm animate-in fade-in duration-300"
    >
      <div 
        onClick={e => e.stopPropagation()}
        className="absolute inset-x-0 bottom-0 top-[15%] bg-white dark:bg-gray-950 rounded-t-[40px] p-8 shadow-2xl animate-in slide-in-from-bottom duration-500 flex flex-col"
      >
        <div className="w-12 h-1 bg-gray-200 dark:bg-gray-800 rounded-full mx-auto mb-8 shrink-0 cursor-grab active:cursor-grabbing" onClick={onClose} />
        
        <div className="flex-1 overflow-y-auto space-y-8 pr-1 custom-scrollbar">
          <div className="text-left">
            <h3 className="text-xl font-bold text-primary dark:text-gray-50 mb-2">Filters<span className="text-accent">.</span></h3>
            <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Customize how you want to access your desired or luxury assets.</p>
          </div>

          
          <div className="space-y-4">
            <label className="text-xs font-bold uppercase tracking-[0.2em] text-gray-400 dark:text-gray-500">Location</label>
            <div className="relative">
              <input 
                type="text" 
                placeholder="City, Country or Region"
                value={localFilters.location}
                onChange={e => setLocalFilters({...localFilters, location: e.target.value})}
                className="w-full px-5 py-4 bg-gray-50 dark:bg-gray-900 rounded-[2rem] border-2 border-transparent dark:border-transparent focus:outline-none focus:border-accent focus:bg-white dark:focus:bg-gray-950 text-sm font-bold text-primary dark:text-gray-50 transition-all placeholder:font-medium placeholder:text-gray-400"
              />
            </div>
          </div>

          <div className="space-y-4">
            <label className="text-xs font-bold uppercase tracking-[0.2em] text-gray-400 dark:text-gray-500">Asset Strategy</label>
            <div className="grid grid-cols-1 gap-2.5">
              {(['Co-own', 'Share', 'Swap'] as ListingGoal[]).map(goal => {
                const isSelected = localFilters.goals.includes(goal);
                return (
                <motion.button
                  key={goal}
                  whileTap={{ scale: 0.98 }}
                  onClick={() => {
                    setLocalFilters(prev => ({
                      ...prev,
                      goals: isSelected 
                        ? prev.goals.filter(g => g !== goal)
                        : [...prev.goals, goal]
                    }));
                  }}
                  className={`w-full flex items-center justify-between px-5 py-5 rounded-[2rem] border-2 transition-all duration-300 ${
                    isSelected
                      ? 'border-accent bg-accent/5 ring-4 ring-accent/5' 
                      : 'border-gray-50 dark:border-gray-800 bg-gray-50 dark:bg-gray-900/50 hover:bg-gray-50 dark:hover:bg-gray-900'
                  }`}
                >
                  <div className="flex items-center gap-4">
                    <div className={`w-12 h-12 rounded-2xl flex items-center justify-center transition-all duration-300 ${
                      isSelected ? 'bg-accent text-white shadow-xl shadow-accent/30' : 'bg-white dark:bg-gray-950 text-blue-500 border border-gray-100 dark:border-gray-800'
                    }`}>
                      {goal === 'Co-own' && <Users className="w-6 h-6" />}
                      {goal === 'Share' && <Handshake className="w-6 h-6" />}
                      {goal === 'Swap' && <RefreshCcw className="w-6 h-6" />}
                    </div>
                    <div className="text-left">
                      <p className={`text-sm font-bold ${isSelected ? 'text-primary dark:text-gray-50' : 'text-gray-500 dark:text-gray-400'}`}>{goal}</p>
                      <p className="text-[10px] text-gray-400 dark:text-gray-500 font-medium max-w-[180px] leading-tight mt-0.5">
                        {goal === 'Co-own' ? 'Build equity through shared ownership' : 
                         goal === 'Share' ? 'Pay for access without the ownership' : 
                         'Swap usage with other luxury assets'}
                      </p>
                    </div>
                  </div>
                  <div className={`w-6 h-6 rounded-full border-2 flex items-center justify-center transition-all duration-500 ${
                    isSelected ? 'bg-accent border-accent scale-110 shadow-lg shadow-accent/20' : 'bg-transparent border-gray-200 dark:border-gray-700 scale-90'
                  }`}>
                    {isSelected && <CheckCircle2 className="w-3.5 h-3.5 text-white" />}
                  </div>
                </motion.button>
              )})}
            </div>
          </div>

          <div className="space-y-4 pt-2">
            <div className="flex items-center justify-between p-5 bg-gray-50 dark:bg-gray-900/50 rounded-[2rem] border-2 border-transparent dark:border-transparent">
              <div className="flex items-center gap-3">
                <div className={`w-12 h-12 rounded-2xl flex items-center justify-center transition-colors ${localFilters.verifiedOnly ? 'bg-green-500 text-white shadow-xl shadow-green-200' : 'bg-white dark:bg-gray-950 text-gray-400 dark:text-gray-500 border border-gray-100 dark:border-gray-800'}`}>
                  <ShieldCheck className="w-6 h-6" />
                </div>
                <div className="text-left">
                  <h4 className="text-sm font-bold text-primary dark:text-gray-50">Verified Only</h4>
                  <p className="text-[10px] text-gray-500 dark:text-gray-400 font-medium">Checked & Authenticated assets</p>
                </div>
              </div>
              <motion.button
                whileTap={{ scale: 0.9 }}
                onClick={() => setLocalFilters({...localFilters, verifiedOnly: !localFilters.verifiedOnly})}
                className={`w-14 h-7 rounded-full relative transition-colors duration-300 ${localFilters.verifiedOnly ? 'bg-green-500' : 'bg-gray-200 dark:bg-gray-800'}`}
              >
                <motion.div 
                  animate={{ x: localFilters.verifiedOnly ? 30 : 2 }}
                  className="absolute top-1 left-0 w-5 h-5 bg-white dark:bg-gray-950 rounded-full shadow-md"
                />
              </motion.button>
            </div>
          </div>

          <div className="space-y-5 pb-8">
            <label className="text-xs font-bold uppercase tracking-[0.2em] text-gray-400 dark:text-gray-500">Price Range (AED)</label>
            <div className="flex gap-4">
              <div className="flex-1 space-y-2">
                <span className="text-[10px] font-bold text-gray-400 dark:text-gray-500 ml-1">MIN PRICE</span>
                <div className="relative">
                  <span className="absolute left-4 top-1/2 -translate-y-1/2 text-primary dark:text-gray-50 font-bold text-xs">AED</span>
                  <input 
                    type="number" 
                    placeholder="0"
                    value={localFilters.minPrice}
                    onChange={e => setLocalFilters({...localFilters, minPrice: e.target.value})}
                    className="w-full pl-12 pr-4 py-4 bg-gray-50 dark:bg-gray-900 rounded-2xl border-2 border-transparent dark:border-transparent focus:outline-none focus:border-accent focus:bg-white dark:focus:bg-gray-950 text-sm font-bold text-primary dark:text-gray-50 transition-all"
                  />
                </div>
              </div>
              <div className="flex-1 space-y-2">
                <span className="text-[10px] font-bold text-gray-400 dark:text-gray-500 ml-1">MAX PRICE</span>
                <div className="relative">
                  <span className="absolute left-4 top-1/2 -translate-y-1/2 text-primary dark:text-gray-50 font-bold text-xs">AED</span>
                  <input 
                    type="number" 
                    placeholder="50,000+"
                    value={localFilters.maxPrice}
                    onChange={e => setLocalFilters({...localFilters, maxPrice: e.target.value})}
                    className="w-full pl-12 pr-4 py-4 bg-gray-50 dark:bg-gray-900 rounded-2xl border-2 border-transparent dark:border-transparent focus:outline-none focus:border-accent focus:bg-white dark:focus:bg-gray-950 text-sm font-bold text-primary dark:text-gray-50 transition-all"
                  />
                </div>
              </div>
            </div>
          </div>
        </div>

        <div className="grid grid-cols-2 gap-4 mt-auto pt-8 pb-10">
          <motion.button 
            whileTap={{ scale: 0.96 }}
            onClick={() => {
              const reset: FilterState = { location: '', minPrice: '', maxPrice: '', goals: [], verifiedOnly: false };
              setLocalFilters(reset);
              onUpdate(reset);
              onClose();
            }}
            className="py-4.5 bg-gray-50 dark:bg-gray-900 text-gray-500 dark:text-gray-400 rounded-[2rem] text-xs font-bold uppercase tracking-[0.2em] hover:text-primary dark:hover:text-gray-50 transition-colors border border-gray-100 dark:border-gray-800"
          >
            Clear All
          </motion.button>
          <motion.button 
            whileTap={{ scale: 0.96 }}
            onClick={() => {
              onUpdate(localFilters);
              onClose();
            }}
            className="py-4.5 bg-primary dark:bg-gray-900 text-white rounded-[2rem] text-xs font-bold uppercase tracking-[0.2em] shadow-2xl shadow-primary/30"
          >
            Show Results
          </motion.button>
        </div>
      </div>
    </div>
  );
};
