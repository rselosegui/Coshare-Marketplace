const ListAssetFlow = ({ onCancel, onComplete, initialGoal = 'Co-own' }: { onCancel: () => void, onComplete: () => void, initialGoal?: ListingGoal }) => {
  const [step, setStep] = useState(1);
  const [form, setForm] = useState<ListingForm>({
    entityType: 'individual',
    category: '1', // Default to Cars
    subcategory: '',
    name: '',
    description: '',
    location: '',
    specifications: {},
    goal: initialGoal,
    isVerified: false
  });

  const selectedCategory = categories.find(c => c.id === form.category);

  const nextStep = () => setStep(s => s + 1);
  const prevStep = () => setStep(s => s - 1);

  return (
    <div className="flex flex-col h-full bg-white dark:bg-gray-950 pb-24 overflow-hidden">
      {/* Top Bar */}
      <div className="pt-8 pb-4 px-4 border-b border-gray-100 dark:border-gray-800 flex items-center bg-white dark:bg-gray-950 sticky top-0 z-10">
        <button onClick={step === 1 ? onCancel : prevStep} className="p-2 -ml-2 text-primary dark:text-gray-50">
          <ChevronLeft className="w-6 h-6" />
        </button>
        <h2 className="flex-1 text-center font-bold text-lg text-primary dark:text-gray-50">List Asset</h2>
        <div className="w-10"></div>
      </div>

      {/* Progress Bar */}
      <div className="flex px-4 pt-4 gap-2">
        {[1, 2, 3].map(i => (
          <div key={i} className={`h-1.5 flex-1 rounded-full transition-colors ${i <= step ? 'bg-primary dark:bg-gray-900' : 'bg-gray-100 dark:bg-gray-800'}`} />
        ))}
      </div>

      <div className="flex-1 overflow-y-auto px-4 py-6">
        {step === 1 && (
          <div className="animate-in fade-in slide-in-from-right-4 duration-300">
            <h3 className="text-xl font-bold text-primary dark:text-gray-50 mb-6">Classification & Category</h3>
            
            <div className="space-y-6">
              <div>
                <label className="text-xs font-bold uppercase tracking-wider text-gray-400 dark:text-gray-500 mb-3 block">Entity Type</label>
                <div className="grid grid-cols-2 gap-3">
                  {(['individual', 'business'] as EntityType[]).map(type => (
                    <motion.button
                      key={type}
                      whileTap={{ scale: 0.95 }}
                      onClick={() => setForm({ ...form, entityType: type })}
                      className={`py-3 px-4 rounded-xl border-2 transition-all font-semibold capitalize ${
                        form.entityType === type 
                          ? 'border-primary bg-primary dark:bg-gray-900/5 text-primary dark:text-gray-50' 
                          : 'border-gray-100 dark:border-gray-800 bg-gray-50 dark:bg-gray-900 text-gray-400 dark:text-gray-500 hover:border-gray-200 dark:border-gray-700 dark:hover:border-gray-700'
                      }`}
                    >
                      {type}
                    </motion.button>
                  ))}
                </div>
              </div>

              <div>
                <label className="text-xs font-bold uppercase tracking-wider text-gray-400 dark:text-gray-500 mb-3 block">Main Category</label>
                <div className="grid grid-cols-3 gap-3">
                  {categories.filter(c => c.id !== '0').map(cat => {
                    const Icon = IconMap[cat.icon];
                    return (
                      <motion.button
                        key={cat.id}
                        whileTap={{ scale: 0.95 }}
                        onClick={() => setForm({ ...form, category: cat.id, subcategory: '' })}
                        className={`flex flex-col items-center gap-2 p-3 rounded-xl border-2 transition-all ${
                          form.category === cat.id 
                            ? 'border-primary bg-primary dark:bg-gray-900/5 text-primary dark:text-gray-50' 
                            : 'border-gray-100 dark:border-gray-800 bg-gray-50 dark:bg-gray-900 text-gray-400 dark:text-gray-500 hover:border-gray-200 dark:border-gray-700 dark:hover:border-gray-700'
                        }`}
                      >
                        <Icon className="w-6 h-6" />
                        <span className="text-[10px] font-bold">{cat.name}</span>
                      </motion.button>
                    )
                  })}
                </div>
              </div>

              {selectedCategory && (
                <div className="animate-in fade-in slide-in-from-bottom-2 duration-300">
                  <label className="text-xs font-bold uppercase tracking-wider text-gray-400 dark:text-gray-500 mb-3 block">Subcategory</label>
                  <div className="grid grid-cols-2 gap-2">
                    {selectedCategory.subcategories.map(sub => (
                      <motion.button
                        key={sub}
                        whileTap={{ scale: 0.95 }}
                        onClick={() => setForm({ ...form, subcategory: sub })}
                        className={`py-2.5 px-4 rounded-xl border font-medium text-xs transition-all ${
                          form.subcategory === sub 
                            ? 'bg-primary dark:bg-gray-900 text-accent border-primary' 
                            : 'bg-white dark:bg-gray-950 border-gray-100 dark:border-gray-800 text-gray-500 dark:text-gray-400 hover:border-gray-300'
                        }`}
                      >
                        {sub}
                      </motion.button>
                    ))}
                  </div>
                </div>
              )}
            </div>
          </div>
        )}

        {step === 2 && (
          <div className="animate-in fade-in slide-in-from-right-4 duration-300 space-y-8">
            <h3 className="text-xl font-bold text-primary dark:text-gray-50">Main Information</h3>
            
            <div className="bg-gray-50 dark:bg-gray-900 rounded-2xl p-4 flex items-center gap-4 border border-gray-100 dark:border-gray-800">
              <div className="w-10 h-10 bg-white dark:bg-gray-950 rounded-full flex items-center justify-center text-primary dark:text-gray-50 shadow-sm">
                {selectedCategory && React.createElement(IconMap[selectedCategory.icon], { className: "w-5 h-5" })}
              </div>
              <div>
                <p className="text-[10px] font-bold tracking-widest text-gray-400 dark:text-gray-500 uppercase">Selected Channel</p>
                <p className="text-sm font-bold text-primary dark:text-gray-50">{form.entityType} • {selectedCategory?.name} • {form.subcategory}</p>
              </div>
            </div>

            <div className="space-y-5">
              <div className="space-y-2">
                <label className="text-xs font-bold text-primary dark:text-gray-50 ml-1">Asset Name</label>
                <input 
                  type="text" 
                  placeholder="e.g. 2024 Porsche 911 Targa"
                  value={form.name}
                  onChange={e => setForm({...form, name: e.target.value})}
                  className="w-full px-4 py-3 bg-gray-50 dark:bg-gray-900 rounded-xl border border-gray-100 dark:border-gray-800 focus:ring-1 focus:ring-accent/20 focus:border-accent outline-none text-sm font-medium"
                />
              </div>

              <div className="space-y-2">
                <label className="text-xs font-bold text-primary dark:text-gray-50 ml-1">Description</label>
                <textarea 
                  rows={3}
                  placeholder="Tell potential cosharers about the asset history, usage, and why co-owning it is operationally efficient."
                  value={form.description}
                  onChange={e => setForm({...form, description: e.target.value})}
                  className="w-full px-4 py-3 bg-gray-50 dark:bg-gray-900 rounded-xl border border-gray-100 dark:border-gray-800 focus:ring-1 focus:ring-accent/20 focus:border-accent outline-none text-sm font-medium resize-none"
                />
              </div>

              <div className="space-y-2">
                <label className="text-xs font-bold text-primary dark:text-gray-50 ml-1">Location</label>
                <div className="relative">
                  <MapPin className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 dark:text-gray-500" />
                  <input 
                    type="text" 
                    placeholder="Dubai, UAE"
                    value={form.location}
                    onChange={e => setForm({...form, location: e.target.value})}
                    className="w-full pl-9 pr-4 py-3 bg-gray-50 dark:bg-gray-900 rounded-xl border border-gray-100 dark:border-gray-800 focus:ring-1 focus:ring-accent/20 focus:border-accent outline-none text-sm font-medium"
                  />
                </div>
              </div>

              <div className="space-y-4">
                <label className="text-xs font-bold text-primary dark:text-gray-50 ml-1 uppercase tracking-widest">Asset Specifications</label>
                <div className="grid grid-cols-2 gap-3">
                  {(categorySpecs[form.category] || ['Model', 'Year', 'Condition']).map(spec => (
                    <div key={spec} className="space-y-1.5 text-left">
                      <label className="text-[10px] font-bold text-gray-400 dark:text-gray-500 uppercase tracking-tight ml-1">{spec}</label>
                      {specOptions[spec] ? (
                        <select
                          value={form.specifications[spec] || ''}
                          onChange={e => setForm({
                            ...form,
                            specifications: { ...form.specifications, [spec]: e.target.value }
                          })}
                          className="w-full px-3 py-2.5 bg-gray-50 dark:bg-gray-900 rounded-xl border border-gray-100 dark:border-gray-800 focus:ring-1 focus:ring-accent/20 focus:border-accent outline-none text-xs font-medium appearance-none"
                        >
                          <option value="">Select {spec}</option>
                          {specOptions[spec].map(opt => (
                            <option key={opt} value={opt}>{opt}</option>
                          ))}
                        </select>
                      ) : (
                        <input 
                          type="text" 
                          placeholder={spec === 'Mileage' ? '0 km' : spec === 'Year' ? '2024' : 'Enter value'}
                          value={form.specifications[spec] || ''}
                          onChange={e => setForm({
                            ...form, 
                            specifications: { ...form.specifications, [spec]: e.target.value }
                          })}
                          className="w-full px-3 py-2.5 bg-gray-50 dark:bg-gray-900 rounded-xl border border-gray-100 dark:border-gray-800 focus:ring-1 focus:ring-accent/20 focus:border-accent outline-none text-xs font-medium"
                        />
                      )}
                    </div>
                  ))}
                </div>
              </div>

              <div className="space-y-3">
                <label className="text-xs font-bold text-primary dark:text-gray-50 ml-1">Listing Goal</label>
                <div className="grid grid-cols-1 gap-2">
                  {(['Co-own', 'Share', 'Swap', 'All'] as ListingGoal[]).map(goal => (
                    <motion.button
                      key={goal}
                      whileTap={{ scale: 0.98 }}
                      onClick={() => setForm({ ...form, goal })}
                      className={`p-4 rounded-2xl border-2 text-left transition-all relative overflow-hidden ${
                        form.goal === goal 
                          ? 'border-primary bg-primary dark:bg-gray-900/5' 
                          : 'border-gray-100 dark:border-gray-800 bg-gray-50 dark:bg-gray-900 hover:border-gray-200 dark:border-gray-700 dark:hover:border-gray-700'
                      }`}
                    >
                      <div className="flex justify-between items-center mb-1">
                        <span className={`text-sm font-black uppercase tracking-tight ${form.goal === goal ? 'text-primary dark:text-gray-50' : 'text-gray-400 dark:text-gray-500'}`}>
                          {goal}
                        </span>
                        {form.goal === goal && <CheckCircle2 className="w-5 h-5 text-accent" />}
                      </div>
                      <p className={`text-[10px] leading-tight font-medium ${form.goal === goal ? 'text-primary dark:text-gray-50/70' : 'text-gray-400 dark:text-gray-500'}`}>
                        {goalDescriptions[goal]}
                      </p>
                    </motion.button>
                  ))}
                </div>
              </div>

              <div className="bg-primary dark:bg-gray-900 rounded-2xl p-4 text-white">
                <div className="flex justify-between items-start mb-2">
                  <div className="flex items-center gap-2">
                    <ShieldCheck className="w-5 h-5 text-accent" />
                    <span className="font-bold text-sm">Boost Visibility</span>
                  </div>
                  <button 
                    onClick={() => setForm({...form, isVerified: !form.isVerified})}
                    className={`w-10 h-5 rounded-full relative transition-colors ${form.isVerified ? 'bg-accent' : 'bg-gray-600'}`}
                  >
                    <div className={`absolute top-1 w-3 h-3 bg-white dark:bg-gray-950 rounded-full transition-all ${form.isVerified ? 'right-1' : 'left-1'}`} />
                  </button>
                </div>
                <p className="text-[10px] opacity-80 leading-relaxed">
                  Is this asset properly insured and maintained by a registered service center? Verified assets get the <span className="text-accent font-bold">Coshare Green Badge</span>.
                </p>
                {form.isVerified && (
                  <div className="mt-3 flex items-center gap-2 bg-white dark:bg-gray-950/10 p-2 rounded-lg border border-white dark:border-gray-950/5 animate-in zoom-in-95 duration-200">
                    <CheckCircle2 className="w-4 h-4 text-accent" />
                    <span className="text-[9px] font-bold uppercase tracking-widest">Green Badge Eligibility Confirmed</span>
                  </div>
                )}
              </div>
            </div>
          </div>
        )}

        {step === 3 && (
          <div className="animate-in fade-in slide-in-from-right-4 duration-300 space-y-8">
            <div className="text-center space-y-2">
              <div className="w-16 h-16 bg-accent-light text-accent rounded-full flex items-center justify-center mx-auto shadow-sm">
                <ShoppingBag className="w-8 h-8" />
              </div>
              <h3 className="text-xl font-bold text-primary dark:text-gray-50 uppercase tracking-tight">Final Review</h3>
              <p className="text-xs text-gray-500 dark:text-gray-400 font-medium tracking-wide">Carefully review your market-ready listing.</p>
            </div>

            <div className="space-y-4">
              <div className="bg-gray-50 dark:bg-gray-900 rounded-2xl p-5 border border-gray-100 dark:border-gray-800 space-y-4">
                <div className="flex justify-between items-start">
                  <div>
                    <h4 className="text-base font-bold text-primary dark:text-gray-50">{form.name || "Unnamed Asset"}</h4>
                    <p className="text-xs font-medium text-gray-500 dark:text-gray-400">{form.subcategory} • {selectedCategory?.name}</p>
                  </div>
                  {form.isVerified && <CheckCircle2 className="w-6 h-6 text-green-500" />}
                </div>
                
                <div className="grid grid-cols-2 gap-4 py-2 border-t border-gray-200 dark:border-gray-700/50">
                  <div className="space-y-1">
                    <p className="text-[9px] font-bold text-gray-400 dark:text-gray-500 tracking-widest uppercase">Entity</p>
                    <p className="text-[11px] font-bold text-primary dark:text-gray-50 capitalize">{form.entityType}</p>
                  </div>
                  <div className="space-y-1">
                    <p className="text-[9px] font-bold text-gray-400 dark:text-gray-500 tracking-widest uppercase">Goal</p>
                    <p className="text-[11px] font-bold text-primary dark:text-gray-50">{form.goal}</p>
                  </div>
                </div>

                {Object.keys(form.specifications).length > 0 && (
                  <div className="pt-4 border-t border-gray-200 dark:border-gray-700/50">
                    <p className="text-[9px] font-bold text-gray-400 dark:text-gray-500 tracking-widest uppercase mb-2">Specifications</p>
                    <div className="grid grid-cols-2 gap-2">
                      {Object.entries(form.specifications).map(([key, value]) => (
                        value && (
                          <div key={key} className="flex justify-between items-center gap-2">
                            <span className="text-[10px] text-gray-500 dark:text-gray-400">{key}:</span>
                            <span className="text-[10px] font-bold text-primary dark:text-gray-50 truncate">{value}</span>
                          </div>
                        )
                      ))}
                    </div>
                  </div>
                )}
              </div>
            </div>
          </div>
        )}
      </div>

      <div className="absolute bottom-0 left-0 right-0 py-2 px-4 bg-white dark:bg-gray-950/90 backdrop-blur-md border-t border-gray-100 dark:border-gray-800 z-30 mx-auto max-w-md w-full">
        <motion.button
          disabled={
            (step === 1 && !form.subcategory) ||
            (step === 2 && (!form.name || !form.location))
          }
          whileTap={(!((step === 1 && !form.subcategory) || (step === 2 && (!form.name || !form.location)))) ? { scale: 0.96 } : {}}
          onClick={step === 3 ? onComplete : nextStep}
          className={`w-full py-4 rounded-2xl font-bold flex items-center justify-center gap-2 transition-all shadow-lg ${
            ((step === 1 && !form.subcategory) || (step === 2 && (!form.name || !form.location)))
              ? 'bg-gray-100 dark:bg-gray-800 text-gray-400 dark:text-gray-500 cursor-not-allowed' 
              : 'bg-primary dark:bg-gray-900 text-accent shadow-primary/20 hover:shadow-primary/30 active:scale-[0.98]'
          }`}
        >
          {step === 3 ? 'Submit for Review' : 'Proceed to Next Step'}
          {step < 3 && <ChevronRight className="w-5 h-5" />}
        </motion.button>
      </div>
    </div>
  )
};
